CREATE TABLE blacklisted_media (
    media_id INT AUTO_INCREMENT PRIMARY KEY,
    image_hash VARCHAR(16) NOT NULL,
    added_by_type VARCHAR(50) NOT NULL DEFAULT 'MANUAL', -- 'MANUAL' or 'AUTOMATED'
    spammer_username VARCHAR(128) NULL,
    spammer_id VARCHAR(64) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX(image_hash)
);



-- Migration: widen image_hash for 16x16 pHash (256 bits = 64 hex chars)
--            and add UNIQUE constraint to prevent race-condition duplicates.
--
-- IMPORTANT: existing rows were hashed with an 8x8 pHash (16 hex chars).
-- Widening the column does NOT retroactively re-hash them. Old rows will
-- simply never match new 64-char hashes via Hamming distance (mismatched
-- lengths -> treated as no-match, see getHammingDistance change below).
-- Effectively this migration resets your blocklist going forward.
-- If you want old entries to keep working, see the backfill note at the
-- bottom of this file.
 
ALTER TABLE blacklisted_media
    MODIFY COLUMN image_hash VARCHAR(64) NOT NULL;
 
-- Prevents two near-simultaneous INSERTs of the same hash from both
-- succeeding (the old check-then-insert pattern in code had a race window).
ALTER TABLE blacklisted_media
    ADD UNIQUE KEY unique_image_hash (image_hash);
